    ```js
    const WHEEL_CIRC_M = Math.PI * 0.06; // must match vel_acc_extraction.py WHEEL_DIAMETER_M
    const SMOOTH_N = 15;

    function computeVelFromSamples(samples) {
      // Filter to valid (magnet ok) samples
      const valid = samples.filter(s => s.magnet_ok === 1);
      if (valid.length < 2) return [];

      // Unwrap angle_counts (handle 0→4095 rollover)
      const unwrapped = [valid[0].angle_counts];
      for (let i = 1; i < valid.length; i++) {
        const delta = valid[i].angle_counts - valid[i - 1].angle_counts;
        // If |delta| > 2048, a rollover occurred
        if (delta > 2048)       unwrapped.push(unwrapped[i - 1] + delta - 4096);
        else if (delta < -2048) unwrapped.push(unwrapped[i - 1] + delta + 4096);
        else                    unwrapped.push(unwrapped[i - 1] + delta);
      }

      // Compute raw velocity for each consecutive pair
      const rawVels = [];
      for (let i = 1; i < valid.length; i++) {
        const dt_s = (valid[i].timestamp_us - valid[i - 1].timestamp_us) / 1_000_000;
        if (dt_s <= 0 || dt_s > 0.1) continue; // skip bad time gaps
        const dCounts = unwrapped[i] - unwrapped[i - 1];
        const dDist_m = (dCounts / 4096) * WHEEL_CIRC_M;
        rawVels.push({ t_s: valid[i].timestamp_us / 1_000_000, v: dDist_m / dt_s });
      }

      // Rolling mean over SMOOTH_N samples
      const smoothed = rawVels.map((pt, idx) => {
        const start = Math.max(0, idx - SMOOTH_N + 1);
        const window = rawVels.slice(start, idx + 1);
        const mean = window.reduce((sum, p) => sum + p.v, 0) / window.length;
        return { t_s: pt.t_s, v: mean };
      });

      return smoothed;
    }
    ```
