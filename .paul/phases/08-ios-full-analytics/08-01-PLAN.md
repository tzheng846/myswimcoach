**3d — Add TimeToXSlider component (above the main export):**
```jsx
function TimeToXSlider({ timeArr, distArr, baselineEndS, headWaistM = 0 }) {
  const [targetM, setTargetM] = React.useState(15);

  const timeToX = React.useMemo(() => {
    if (!timeArr?.length || !distArr?.length || baselineEndS == null) return null;
    const baseIdx = timeArr.findIndex(t => t >= baselineEndS);
    if (baseIdx < 0) return null;
    const distBase = distArr[baseIdx];
    const waistTarget = targetM - headWaistM;
    if (waistTarget <= 0) return null;
    const crossIdx = distArr.findIndex((d, i) => i >= baseIdx && d >= distBase + waistTarget);
    if (crossIdx < 0) return null;
    return (timeArr[crossIdx] - timeArr[baseIdx]).toFixed(2);
  }, [timeArr, distArr, baselineEndS, headWaistM, targetM]);

  return (
    <View>
      <Text style={styles.metricValue}>{timeToX != null ? `${timeToX} s` : '--'}</Text>
      <Text style={[styles.metricUnit, { marginBottom: 8 }]}>to {targetM} m</Text>
      <Slider
        minimumValue={1}
        maximumValue={25}
        step={1}
        value={targetM}
        onValueChange={setTargetM}
        minimumTrackTintColor="#1E3A5F"
        maximumTrackTintColor="#555"
        thumbTintColor="#1E3A5F"
      />
      <View style={styles.sliderLabels}>
        <Text style={styles.sliderLabel}>1 m</Text>
        <Text style={styles.sliderLabel}>25 m</Text>
      </View>
    </View>
  );
}
```
